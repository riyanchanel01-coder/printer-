        function syntaxHighlightJson(json) {
            json = json.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
            return json.replace(/("(\\u[a-zA-Z0-9]{4}|\\[^u]|[^\\"])*"(\s*:)?|\b(true|false|null)\b|-?\d+(?:\.\d*)?(?:[eE][+\-]?\d+)?)/g, function (match) {
                let cls = 'text-amber-300';
                if (/^"/.test(match)) {
                    if (/:$/.test(match)) {
                        cls = 'text-sky-400 font-semibold';
                    } else {
                        cls = 'text-emerald-300';
                    }
                } else if (/true|false/.test(match)) {
                    cls = 'text-purple-400';
                } else if (/null/.test(match)) {
                    cls = 'text-red-400';
                }
                return '<span class="' + cls + '">' + match + '</span>';
            });
        }

        async function connectBluetoothPrinter() {
            try {
                if (!navigator.bluetooth) {
                    showToast("Tidak Didukung", "Web Bluetooth API tidak didukung di browser ini. Gunakan Chrome atau Edge.", "error");
                    return;
                }

                showToast("Memindai", "Mencari perangkat printer thermal Bluetooth...", "info");

                bluetoothDevice = await navigator.bluetooth.requestDevice({
                    acceptAllDevices: true,
                    optionalServices: [
                        '000018f0-0000-1000-8000-00805f9b34fb', 
                        '0000ff00-0000-1000-8000-00805f9b34fb',
                        'e7810a71-73ae-499d-8c15-faa9aef0c3f2',
                        '49535343-fe7d-4ae5-8fa9-9fafd205e455'
                    ]
                });

                const server = await bluetoothDevice.gatt.connect();
                const services = await server.getPrimaryServices();
                let targetCharacteristic = null;

                for (const service of services) {
                    try {
                        const characteristics = await service.getCharacteristics();
                        targetCharacteristic = characteristics.find(c => c.properties.write || c.properties.writeWithoutResponse);
                        if (targetCharacteristic) break;
                    } catch (err) {
                        console.warn("Lewati service yang dibatasi:", err);
                    }
                }

                if (!targetCharacteristic) {
                    throw new Error("Karakteristik untuk menulis data ESC/POS tidak ditemukan pada printer ini.");
                }

                bluetoothCharacteristic = targetCharacteristic;

                const badge = document.getElementById('bluetooth-status-badge');
                badge.className = "inline-flex items-center px-2.5 py-1 rounded-full text-xs font-medium bg-emerald-100 text-emerald-800 border border-emerald-200";
                badge.innerHTML = `<span class="w-2 h-2 mr-1.5 bg-emerald-500 rounded-full"></span> Terhubung: ${bluetoothDevice.name || 'Thermal Printer'}`;
                
                document.getElementById('bt-connect-btn').innerText = "Terhubung";
                showToast("Berhasil Terhubung", `Terhubung ke printer ${bluetoothDevice.name || 'Bluetooth'}`, "success");

            } catch (error) {
                console.error(error);
                showToast("Koneksi Gagal", "Gagal menyambungkan perangkat Bluetooth atau dibatalkan.", "error");
            }
        }

        async function executeBluetoothPrint() {
            if (!currentParsedData) {
                showToast("Peringatan", "Belum ada data untuk dicetak.", "error");
                return;
            }

            if (!bluetoothCharacteristic) {
                showToast("Belum Terhubung", "Harap hubungkan printer Bluetooth terlebih dahulu.", "error");
                return;
            }

            try {
                let encoder = new TextEncoder();
                let bytes = [];

                bytes.push(0x1B, 0x40);
                bytes.push(0x1B, 0x61, 0x01);
                bytes.push(0x1B, 0x45, 0x01);
                bytes.push(...encoder.encode("BUKTI TRANSFER VALID\n"));
                bytes.push(0x1B, 0x45, 0x00);
                bytes.push(...encoder.encode(`BANK ${currentParsedData.bank}\n`));
                bytes.push(...encoder.encode(`--------------------------------\n`));

                bytes.push(0x1B, 0x61, 0x00);
                bytes.push(...encoder.encode(`Status   : ${currentParsedData.status}\n`));
                bytes.push(...encoder.encode(`Tanggal  : ${currentParsedData.dateTime}\n`));
                bytes.push(...encoder.encode(`Ref ID   : ${currentParsedData.refId}\n`));
                bytes.push(...encoder.encode(`--------------------------------\n`));

                bytes.push(...encoder.encode(`Penerima :\n`));
                bytes.push(0x1B, 0x45, 0x01);
                bytes.push(...encoder.encode(`${currentParsedData.recipientName}\n`));
                bytes.push(0x1B, 0x45, 0x00);
                bytes.push(...encoder.encode(`No. Rek  : ${currentParsedData.accountNumber}\n`));
                bytes.push(...encoder.encode(`--------------------------------\n`));

                bytes.push(0x1B, 0x61, 0x01);
                bytes.push(0x1D, 0x21, 0x11);
                bytes.push(...encoder.encode(`${currentParsedData.amount}\n`));
                bytes.push(0x1D, 0x21, 0x00);
                bytes.push(...encoder.encode(`--------------------------------\n`));

                bytes.push(...encoder.encode("Terima Kasih Telah Bertransaksi\n\n\n"));

                const chunksize = 512;
                const dataArray = new Uint8Array(bytes);
                for (let i = 0; i < dataArray.length; i += chunksize) {
                    let chunk = dataArray.slice(i, i + chunksize);
                    await bluetoothCharacteristic.writeValue(chunk);
                }

                showToast("Cetak Berhasil", "Struk berhasil dikirim ke printer thermal Bluetooth!", "success");

            } catch (error) {
                console.error(error);
                showToast("Gagal Cetak", "Terjadi kesalahan saat mengirim perintah ke printer.", "error");
            }
        }

        function showToast(title, message, type = 'success') {
            const toast = document.getElementById('toast');
            const toastTitle = document.getElementById('toast-title');
            const toastMessage = document.getElementById('toast-message');
            const toastIcon = document.getElementById('toast-icon');

            toastTitle.innerText = title;
            toastMessage.innerText = message;

            if (type === 'success') {
                toastIcon.innerHTML = '<i class="fa-solid fa-circle-check text-emerald-400"></i>';
            } else if (type === 'error') {
                toastIcon.innerHTML = '<i class="fa-solid fa-circle-exclamation text-red-400"></i>';
            } else {
                toastIcon.innerHTML = '<i class="fa-solid fa-circle-info text-sky-400"></i>';
            }

            toast.classList.remove('translate-y-24', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-24', 'opacity-0');
            }, 3500);
        }
